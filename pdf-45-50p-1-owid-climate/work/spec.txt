//  lib/analysis/ndc-gap.ts   interface  GapResult  {    onTrack:  boolean;    gapKt:  number;                     //  양수 = 초과 ,  음수 = 달성  예상    ambitionGapFactor:  number;         //  필요  감축속도  /  현재  감축속도    requiredAnnualReduction:  number;    currentAnnualReduction:  number;    projectedTimeseries:  {  year:  number;  value:  number  }[];    targetLine:  {  year:  number;  value:  number  }[];  }   export  function  calculateNdcGap(    emissions:  {  year:  number;  value:  number  }[],    ndc:  {      base_year:  number;      base_year_emissions:  number;      target_year:  number;      reduction_pct:  number;    }  ):  GapResult  {    //  정렬    const  sorted  =  [...emissions].sort((a,  b)  =>  a.year  -  b.year);    const  latest  =  sorted[sorted.length  -  1];       //  최근  5 년  평균  연간  변화    const  recentYears  =  sorted.filter(e  =>  e.year  >=  latest.year  -  5);    let  avgAnnualChange  =  0;    if  (recentYears.length  >=  2)  {      const  changes:  number[]  =  [];      for  (let  i  =  1;  i  <  recentYears.length;  i++)  {        changes.push(          (recentYears[i].value  -  recentYears[i  -  1].value)  /          (recentYears[i].year  -  recentYears[i  -  1].year)        );

    }      avgAnnualChange  =  changes.reduce((a,  b)  =>  a  +  b,  0)  /  changes.length;    }       //  목표  배출량    const  targetEmissions  =  ndc.base_year_emissions  *  (1  -  ndc.reduction_pct  /  100);       //  선형  외삽    const  yearsToTarget  =  ndc.target_year  -  latest.year;    const  projectedValue  =  latest.value  +  avgAnnualChange  *  yearsToTarget;       //  격차    const  gap  =  projectedValue  -  targetEmissions;       //  필요  연간  감축량    const  requiredAnnual  =  (latest.value  -  targetEmissions)  /  yearsToTarget;       //  야망  격차  배수  ( 현재  속도  대비  필요  속도 )    const  ambitionFactor  =  avgAnnualChange  <  0      ?  requiredAnnual  /  Math.abs(avgAnnualChange)      :  Infinity;   //  배출이  증가  중이면  무한대       //  전망  시계열  생성    const  projected:  {  year:  number;  value:  number  }[]  =  [];    for  (let  y  =  latest.year;  y  <=  ndc.target_year;  y++)  {      projected.push({        year:  y,        value:  Math.max(0,  latest.value  +  avgAnnualChange  *  (y  -  latest.year)),      });    }       //  목표  직선    const  targetLine  =  [      {  year:  ndc.base_year,  value:  ndc.base_year_emissions  },      {  year:  ndc.target_year,  value:  targetEmissions  },    ];     return  {      onTrack:  gap  <=  0,      gapKt:  Math.round(gap),      ambitionGapFactor:  Math.round(ambitionFactor  *  100)  /  100,      requiredAnnualReduction:  Math.round(requiredAnnual),      currentAnnualReduction:  Math.round(avgAnnualChange),      projectedTimeseries:  projected,      targetLine,    };  }
